import librosa
import numpy as np

def extract_audio_features(audio_file, start, end):
    """提取音频特征"""
    # 加载音频片段
    y, sr = librosa.load(audio_file, offset=start, duration=end-start)
    
    features = {
        # 1. 音高特征
        'pitch': librosa.yin(y, fmin=50, fmax=2000),
        
        # 2. 节奏特征
        'tempo': librosa.beat.tempo(y=y, sr=sr)[0],
        
        # 3. 音色特征（MFCC）
        'mfcc': librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13),
        
        # 4. 能量特征
        'energy': librosa.feature.rms(y=y)[0],
        
        # 5. 频谱特征
        'spectral_centroid': librosa.feature.spectral_centroid(y=y, sr=sr)[0],
        
        # 6. 和声特征
        'chroma': librosa.feature.chroma_stft(y=y, sr=sr)
    }
    
    return features

    if len(frames) > 100:
        step = len(frames) // 100
        frames = frames[::step]

    resized_frames = []
    for frame in frames:
        img = Image.fromarray(frame)

class MultiAPIRecognizer:
    """多API联合识别器"""
    
    def __init__(self):
        self.apis = {
            'shazam': ShazamAPI(),
            'acrcloud': ACRCloudAPI(),
            'audd': AudDAPI(),
        }
    
    def recognize_with_fallback(self, audio_path):
        """级联识别策略"""
        results = []
        
        # 1. 优先Shazam（最快）
        result = self.apis['shazam'].recognize(audio_path)
        if result:
            results.append(('shazam', result, 0.95))
        
        # 2. 失败则尝试ACRCloud（最准）
        if not result:
            result = self.apis['acrcloud'].recognize(audio_path)
            if result:
                results.append(('acrcloud', result, 0.98))
        
        # 3. 再失败则尝试AudD（免费）
        if not result:
            result = self.apis['audd'].recognize(audio_path)
            if result:
                results.append(('audd', result, 0.85))
        
        # 4. 交叉验证
        if len(results) > 1:
            return self._cross_validate(results)
        
        return results[0] if results else None
    
    def _cross_validate(self, results):
        """交叉验证多个结果"""
        # 比较歌名相似度
        titles = [r[1]['title'] for r in results]
        
        # 如果多数一致，返回置信度最高的
        from difflib import SequenceMatcher
        
        # 简化：返回置信度最高的
        return max(results, key=lambda x: x[2])

def select_best_segment(song_segments):
    """选择最佳识别片段"""
    
    scores = []
    for seg in song_segments:
        score = 0
        
        # 1. 时长评分（30-45秒最佳）
        duration = seg['end'] - seg['start']
        if 30 <= duration <= 45:
            score += 10
        elif 15 <= duration < 30:
            score += 5
        
        # 2. 能量评分（避免静音段）
        if seg.get('energy', 0) > 0.3:
            score += 5
        
        # 3. 位置评分（中间段更好）
        total_duration = song_segments[-1]['end'] - song_segments[0]['start']
        seg_position = (seg['start'] - song_segments[0]['start']) / total_duration
        if 0.3 <= seg_position <= 0.7:
            score += 5
        
        scores.append((score, seg))
    
    # 返回得分最高的片段
    return max(scores, key=lambda x: x[0])[1]

def normalize_song_name(song_name):
    """歌名标准化"""
    
    # 1. 英文→中文
    if song_name in SONG_NAME_MAPPING:
        song_name = SONG_NAME_MAPPING[song_name]
    
    # 2. 移除特殊标记
    import re
    song_name = re.sub(r'\(.*?\)', '', song_name)
    song_name = re.sub(r'（.*?）', '', song_name)
    song_name = song_name.strip()
    
    # 3. 繁简转换
    from opencc import OpenCC
    cc = OpenCC('t2s')  # 繁体转简体
    song_name = cc.convert(song_name)
    
    return song_name

def search_lyrics_with_normalization(song_name, artist):
    """使用标准化歌名搜索歌词"""
    
    # 原始歌名搜索
    lyrics = search_lyrics(song_name, artist)
    if lyrics:
        return lyrics
    
    # 标准化后搜索
    normalized_name = normalize_song_name(song_name)
    if normalized_name != song_name:
        lyrics = search_lyrics(normalized_name, artist)
        if lyrics:

    # 移除歌手名再搜索（有些歌不需要歌手）

            'density': count
        })
    
    # 2. 找出高密度时段
    high_density_moments = [
        d for d in density_timeline 
        if d['density'] > np.mean([d['density'] for d in density_timeline]) + np.std([d['density'] for d in density_timeline])
    ]

### 补充4：表情包适配抖音要求 ⚠️

**现状**：有 `测试抖音表情包规格.py`，但不完整

**完整适配方案**：

创建新文件：`抖音表情包完整适配.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
抖音表情包完整适配工具

严格按照抖音官方规范：
- GIF大小：≤1MB（推荐500KB）
- 尺寸：240x240 / 480x480 / 720x720
- 时长：≤5秒
- 帧数：≤100帧
- 帧率：10-30fps
"""

import os
from PIL import Image
import imageio
import numpy as np

class DouyinEmojiAdapter:
    """抖音表情包适配器"""
    
    SPECS = {
        'max_file_size': 1024 * 1024,      # 1MB
        'recommended_size': 500 * 1024,     # 500KB
        'dimensions': [(240, 240), (480, 480), (720, 720)],
        'max_duration': 5.0,
        'max_frames': 100,
        'fps_range': (10, 30),
    }
    
    def adapt_gif(self, input_path, output_dir):
        """适配GIF到抖音规格"""
        
        # 读取原始GIF
        frames = imageio.mimread(input_path)
        
        results = {}
        
        for size in self.SPECS['dimensions']:
            output_path = os.path.join(
                output_dir,
                f"{os.path.basename(input_path).split('.')[0]}_{size[0]}x{size[1]}.gif"
            )
            
            # 1. 控制帧数
            if len(frames) > self.SPECS['max_frames']:
                step = len(frames) // self.SPECS['max_frames']
                selected_frames = frames[::step]
            else:
                selected_frames = frames
            
            # 2. 调整尺寸
            resized_frames = []
            for frame in selected_frames:
                img = Image.fromarray(frame)
                img = img.resize(size, Image.Resampling.LANCZOS)
                resized_frames.append(np.array(img))
            
            # 3. 颜色量化（减小文件大小）
            quantized_frames = self._quantize_frames(resized_frames)
            
            # 4. 生成GIF
            duration = self.SPECS['max_duration'] / len(quantized_frames)
            duration = max(duration, 1/30)  # 不超过30fps
            
            imageio.mimsave(
                output_path,
                quantized_frames,
                duration=duration,
                loop=0
            )
            
            # 5. 检查文件大小
            file_size = os.path.getsize(output_path)
            
            # 6. 如果过大，进一步压缩
            if file_size > self.SPECS['max_file_size']:
                output_path = self._compress_further(output_path, size)
                file_size = os.path.getsize(output_path)
            
            # 7. 验证
            is_valid = self._validate(output_path)
            
            results[f"{size[0]}x{size[1]}"] = {
                'path': output_path,
                'size': file_size,
                'size_mb': round(file_size / 1024 / 1024, 2),
                'valid': is_valid,
                'status': '✓' if is_valid else '✗ 超过1MB限制'
            }
        
        return results
    
    def _quantize_frames(self, frames, colors=128):
        """颜色量化"""
        quantized = []
        
        for frame in frames:
            img = Image.fromarray(frame)
            # 转换为调色板模式
            img = img.convert('P', palette=Image.ADAPTIVE, colors=colors)
            quantized.append(img)
        
        return quantized
    
    def _compress_further(self, gif_path, size):
        """进一步压缩"""
        from PIL import Image
        
        # 降低颜色数
        frames = imageio.mimread(gif_path)
        
        # 更激进的颜色量化
        compressed_frames = self._quantize_frames(frames, colors=64)
        
        # 重新保存
        imageio.mimsave(gif_path, compressed_frames, duration=0.1, loop=0)
        
        return gif_path
    
    def _validate(self, gif_path):
        """验证是否符合抖音规格"""
        
        # 检查文件大小
        file_size = os.path.getsize(gif_path)
        if file_size > self.SPECS['max_file_size']:
            return False
        
        # 检查帧数
        frames = imageio.mimread(gif_path)
        if len(frames) > self.SPECS['max_frames']:
            return False
        
        # 检查尺寸
        img = Image.open(gif_path)
        if img.size not in self.SPECS['dimensions']:
            return False
        
        return True
    
    def batch_adapt(self, input_dir, output_dir):
        """批量适配"""
        
        os.makedirs(output_dir, exist_ok=True)
        
        gif_files = [f for f in os.listdir(input_dir) if f.endswith('.gif')]
        
        print(f"找到 {len(gif_files)} 个GIF文件")
        print()
        
        all_results = {}
        
        for gif_file in gif_files:
            input_path = os.path.join(input_dir, gif_file)
            print(f"处理: {gif_file}")
            
            results = self.adapt_gif(input_path, output_dir)
            
            for size, result in results.items():
                print(f"  {size}: {result['size_mb']}MB {result['status']}")
            
            all_results[gif_file] = results
            print()
        
        return all_results

def main():
    """主函数"""
    print("=" * 70)
    print("抖音表情包完整适配工具")
    print("=" * 70)
    print()
    
    input_dir = os.path.expanduser("~/shanshan_materials/表情包素材库")
    output_dir = os.path.expanduser("~/shanshan_materials/表情包_抖音适配版")
    
    adapter = DouyinEmojiAdapter()
    results = adapter.batch_adapt(input_dir, output_dir)
    
    print("=" * 70)
    print("✅ 适配完成！")
    print("=" * 70)
    print()
    print(f"输出目录: {output_dir}")
    print(f"处理文件: {len(results)} 个")
    print()

if __name__ == "__main__":
    main()
```
