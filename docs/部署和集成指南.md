### 示例Schema (schema.sql)

```sql
-- 视频表
CREATE TABLE videos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    duration INT NOT NULL,
    file_size BIGINT NOT NULL,
    resolution VARCHAR(20),
    upload_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    view_count INT DEFAULT 0,
    download_count INT DEFAULT 0,
    category ENUM('full', 'highlight', 'song') NOT NULL,
    status ENUM('active', 'archived') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_category (category),
    INDEX idx_status (status)
);

-- 歌曲表
CREATE TABLE songs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    video_start FLOAT NOT NULL,
    video_end FLOAT NOT NULL,
    duration FLOAT NOT NULL,
    audio_file VARCHAR(500),
    video_file VARCHAR(500),
    lyrics_srt VARCHAR(500),
    lyrics_ass VARCHAR(500),
    play_count INT DEFAULT 0,
    download_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_title (title)
);

-- 表情包表
CREATE TABLE emojis (
    id INT PRIMARY KEY AUTO_INCREMENT,
    category ENUM('dance', 'thanks', 'gift_effect') NOT NULL,
    file_type ENUM('gif', 'png', 'jpg') NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    file_size INT NOT NULL,
    width INT NOT NULL,
    height INT NOT NULL,
    duration FLOAT,
    source_time FLOAT,
    view_count INT DEFAULT 0,
    download_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_category (category)
);

-- 照片表
CREATE TABLE photos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    category ENUM('interval', 'scene_change', 'song', 'gift') NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    file_size INT NOT NULL,
    width INT NOT NULL,
    height INT NOT NULL,
    source_time FLOAT,
    view_count INT DEFAULT 0,
    download_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_category (category),
    INDEX idx_source_time (source_time)
);

-- 统计表
CREATE TABLE statistics (
    id INT PRIMARY KEY AUTO_INCREMENT,
    date DATE UNIQUE NOT NULL,
    total_visits INT DEFAULT 0,
    total_downloads INT DEFAULT 0,
    popular_song_id INT,
    popular_emoji_id INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

## 🔌 后端API部署

### 方案A：Python FastAPI

#### 1. 安装依赖

```bash
cd /var/www/shanshan/backend

# 创建虚拟环境
python3 -m venv venv
source venv/bin/activate

# 安装依赖
pip install fastapi uvicorn sqlalchemy pymysql redis aiofiles python-multipart
pip install -r requirements.txt  # 如有
```

#### 2. 配置文件 (config.py)

```python
# /var/www/shanshan/backend/config.py
import os

class Config:
    # 数据库
    DB_HOST = os.getenv("DB_HOST", "localhost")
    DB_USER = os.getenv("DB_USER", "shanshan")
    DB_PASSWORD = os.getenv("DB_PASSWORD", "your_password")
    DB_NAME = os.getenv("DB_NAME", "shanshan_db")
    
    # Redis
    REDIS_HOST = os.getenv("REDIS_HOST", "localhost")
    REDIS_PORT = int(os.getenv("REDIS_PORT", 6379))
    
    # 文件路径
    MEDIA_ROOT = "/var/www/shanshan/media"
    MATERIAL_ROOT = os.path.expanduser("~/shanshan_materials")
    
    # API
    API_PREFIX = "/api/v1"
    SECRET_KEY = os.getenv("SECRET_KEY", "your_secret_key_here")
    
    # CORS
    CORS_ORIGINS = ["http://localhost:3000", "https://shanshan.com"]
```

#### 3. 主程序 (main.py)

```python
# /var/www/shanshan/backend/main.py
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles
import uvicorn

from config import Config

app = FastAPI(title="观山素材API", version="1.0")

# CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=Config.CORS_ORIGINS,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 静态文件
app.mount("/media", StaticFiles(directory=Config.MEDIA_ROOT), name="media")

# 路由
from routers import videos, songs, emojis, photos, stats

app.include_router(videos.router, prefix=f"{Config.API_PREFIX}/videos", tags=["videos"])
app.include_router(songs.router, prefix=f"{Config.API_PREFIX}/songs", tags=["songs"])
app.include_router(emojis.router, prefix=f"{Config.API_PREFIX}/emojis", tags=["emojis"])
app.include_router(photos.router, prefix=f"{Config.API_PREFIX}/photos", tags=["photos"])
app.include_router(stats.router, prefix=f"{Config.API_PREFIX}/stats", tags=["stats"])

@app.get("/")
def read_root():
    return {"message": "观山素材API v1.0"}

@app.get("/health")
def health_check():
    return {"status": "healthy"}

if __name__ == "__main__":
    uvicorn.run("main:app", host="0.0.0.0", port=8000, reload=False)
```

## 🌐 Nginx配置

### 1. 主配置文件

```nginx
# /etc/nginx/sites-available/shanshan.conf

# API服务
upstream shanshan_api {
    server 127.0.0.1:8000;
}

# 主站
server {
    listen 80;
    server_name shanshan.com www.shanshan.com;
    
    # 重定向到HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name shanshan.com www.shanshan.com;
    
    # SSL证书
    ssl_certificate /etc/letsencrypt/live/shanshan.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/shanshan.com/privkey.pem;
    
    # SSL配置
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    
    # 日志
    access_log /var/log/nginx/shanshan_access.log;
    error_log /var/log/nginx/shanshan_error.log;
    
    # 前端静态文件
    root /var/www/shanshan/frontend/dist;
    index index.html;
    
    # 前端路由
    location / {
        try_files $uri $uri/ /index.html;
    }
    
    # API代理
    location /api/ {
        proxy_pass http://shanshan_api;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    
    # 媒体文件（启用缓存）
    location /media/ {
        alias /var/www/shanshan/media/;
        expires 7d;
        add_header Cache-Control "public, immutable";
    }
    
    # 文件大小限制
    client_max_body_size 100M;
}

# 视频子站
server {
    listen 443 ssl http2;
    server_name video.shanshan.com;
    
    ssl_certificate /etc/letsencrypt/live/shanshan.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/shanshan.com/privkey.pem;
    
    root /var/www/shanshan/frontend/video;
    index index.html;
    
    location / {
        try_files $uri $uri/ /index.html;
    }
    
    location /api/ {
        proxy_pass http://shanshan_api;
    }
    
    location /media/ {
        alias /var/www/shanshan/media/;
    }
}

# 歌曲子站
server {
    listen 443 ssl http2;
    server_name music.shanshan.com;
    
    # 类似配置...
}
```

## 📂 素材同步

### 1. 手动同步

```bash
# 从素材服务器同步到网站媒体目录
rsync -avz --progress \
  ~/shanshan_materials/ \
  /var/www/shanshan/media/

# 或使用已有的同步脚本
bash ~/DouyinLiveRecorder/同步到storage定时任务.sh
```

### 2. 自动同步脚本

```bash
#!/bin/bash
# /var/www/shanshan/scripts/sync_materials.sh

SOURCE_DIR="$HOME/shanshan_materials"
TARGET_DIR="/var/www/shanshan/media"
LOG_FILE="/var/www/shanshan/logs/sync.log"

echo "=== $(date '+%Y-%m-%d %H:%M:%S') 开始同步 ===" >> "$LOG_FILE"

# 同步视频
rsync -avz --progress \
  "$SOURCE_DIR/downloads/" \
  "$TARGET_DIR/videos/" \
  >> "$LOG_FILE" 2>&1

# 同步歌曲音频
rsync -avz --progress \
  "$SOURCE_DIR/歌曲素材库/音频/" \
  "$TARGET_DIR/audio/" \
  >> "$LOG_FILE" 2>&1

# 同步表情包
rsync -avz --progress \
  "$SOURCE_DIR/表情包素材库/" \
  "$TARGET_DIR/emojis/" \
  >> "$LOG_FILE" 2>&1

# 同步截图
rsync -avz --progress \
  "$SOURCE_DIR/关键帧截图/" \
  "$TARGET_DIR/photos/" \
  >> "$LOG_FILE" 2>&1

# 同步字幕
rsync -avz --progress \
  "$SOURCE_DIR/歌词字幕/" \
  "$TARGET_DIR/subtitles/" \
  >> "$LOG_FILE" 2>&1

# 同步元数据
rsync -avz --progress \
  "$SOURCE_DIR/"*.json \
  "$TARGET_DIR/metadata/" \
  >> "$LOG_FILE" 2>&1

# 设置权限
chmod -R 755 "$TARGET_DIR"
chown -R www-data:www-data "$TARGET_DIR"

echo "=== $(date '+%Y-%m-%d %H:%M:%S') 同步完成 ===" >> "$LOG_FILE"
```

### 3. 定时任务

```bash
# 编辑crontab
crontab -e

# 每小时同步一次
0 * * * * /var/www/shanshan/scripts/sync_materials.sh

# 每天凌晨2点全量同步
0 2 * * * /var/www/shanshan/scripts/sync_materials.sh --full
```

## 🗃️ 数据导入

### 1. 导入歌曲数据

```python
# /var/www/shanshan/scripts/import_songs.py
import json
import mysql.connector

# 连接数据库
conn = mysql.connector.connect(
    host="localhost",
    user="shanshan",
    password="your_password",
    database="shanshan_db"
)
cursor = conn.cursor()

# 读取歌曲列表
with open("/var/www/shanshan/media/metadata/歌曲列表.json", "r", encoding="utf-8") as f:
    data = json.load(f)

# 插入数据
for song in data["songs"]:
    cursor.execute("""
        INSERT INTO songs (title, video_start, video_end, duration, 
                          audio_file, video_file, lyrics_srt, lyrics_ass)
        VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
    """, (
        song["title"],
        song["start"],
        song["end"],
        song["duration"],
        f"/media/audio/mp3/{song['id']:03d}_{song['title']}.mp3",
        f"/media/videos/songs/{song['id']:03d}_{song['title']}.mp4",
        f"/media/subtitles/srt/{song['id']:03d}_{song['title']}.srt",
        f"/media/subtitles/ass/{song['id']:03d}_{song['title']}.ass"
    ))

conn.commit()
print(f"✓ 导入了 {len(data['songs'])} 首歌曲")

cursor.close()
conn.close()
```

### 1. 数据库备份

```bash
#!/bin/bash
# /var/www/shanshan/scripts/backup_db.sh

BACKUP_DIR="/var/backups/shanshan"
DATE=$(date +%Y%m%d_%H%M%S)

mkdir -p "$BACKUP_DIR"

mysqldump -u shanshan -p'your_password' shanshan_db \
  | gzip > "$BACKUP_DIR/shanshan_db_$DATE.sql.gz"

# 保留最近7天的备份
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +7 -delete

echo "✓ 数据库备份完成: shanshan_db_$DATE.sql.gz"
```
